cmake_minimum_required(VERSION 3.20)
project(physarum_networks VERSION 0.1.0 LANGUAGES CXX)
find_package(Eigen3 REQUIRED NO_MODULE)
add_library(physarum src/Graph.cpp src/FlowSolver.cpp src/PhysarumModel.cpp)
target_include_directories(physarum PUBLIC include)
target_link_libraries(physarum PUBLIC Eigen3::Eigen)
target_compile_features(physarum PUBLIC cxx_std_20)
set_target_properties(physarum PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(physarum PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(physarum_run src/main.cpp)
target_link_libraries(physarum_run PRIVATE physarum)
target_compile_definitions(physarum_run PRIVATE PHYSARUM_CMAKE_VERSION="${CMAKE_VERSION}")
include(CTest)
if(BUILD_TESTING)
  add_executable(physarum_tests tests/test_main.cpp)
  target_link_libraries(physarum_tests PRIVATE physarum)
  foreach(case IN ITEMS series parallel diamond reverse zero components disconnected near_zero invalid euler positivity timestep conditioning terminals)
    add_test(NAME ${case} COMMAND physarum_tests ${case})
  endforeach()
  add_test(NAME cli_smoke COMMAND physarum_run ${PROJECT_SOURCE_DIR}/experiments/configs/two_routes.cfg ${PROJECT_BINARY_DIR}/smoke)
endif()
