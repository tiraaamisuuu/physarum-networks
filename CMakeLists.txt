cmake_minimum_required(VERSION 3.20)
project(physarum_networks VERSION 0.1.0 LANGUAGES CXX)
find_package(Eigen3 REQUIRED NO_MODULE)
add_library(physarum src/Graph.cpp src/FlowSolver.cpp src/PhysarumModel.cpp src/GrowthModel.cpp src/SparseFlowSolver.cpp src/BiologyModel.cpp)
target_include_directories(physarum PUBLIC include)
target_link_libraries(physarum PUBLIC Eigen3::Eigen)
target_compile_features(physarum PUBLIC cxx_std_20)
set_target_properties(physarum PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(physarum PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(physarum_run src/main.cpp)
target_link_libraries(physarum_run PRIVATE physarum)
target_compile_definitions(physarum_run PRIVATE PHYSARUM_CMAKE_VERSION="${CMAKE_VERSION}")
add_executable(physarum_growth src/growth_main.cpp)
target_link_libraries(physarum_growth PRIVATE physarum)
add_executable(physarum_mesh src/mesh_main.cpp)
target_link_libraries(physarum_mesh PRIVATE physarum)
# Identify the source actually compiled into the mesh executable, independently of Git state.
file(GLOB ENGINE_SOURCE_FILES CONFIGURE_DEPENDS
  "${PROJECT_SOURCE_DIR}/src/*.cpp" "${PROJECT_SOURCE_DIR}/include/physarum/*.hpp")
list(APPEND ENGINE_SOURCE_FILES "${PROJECT_SOURCE_DIR}/CMakeLists.txt")
list(SORT ENGINE_SOURCE_FILES)
set(ENGINE_SOURCE_MATERIAL "")
foreach(source_file IN LISTS ENGINE_SOURCE_FILES)
  file(RELATIVE_PATH source_relative "${PROJECT_SOURCE_DIR}" "${source_file}")
  file(SHA256 "${source_file}" source_digest)
  string(APPEND ENGINE_SOURCE_MATERIAL "${source_relative}=${source_digest}\n")
endforeach()
string(SHA256 ENGINE_SOURCE_FINGERPRINT "${ENGINE_SOURCE_MATERIAL}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ENGINE_SOURCE_FILES})
target_compile_definitions(physarum_mesh PRIVATE PHYSARUM_SOURCE_FINGERPRINT="${ENGINE_SOURCE_FINGERPRINT}")
add_executable(physarum_biology src/biology_main.cpp)
target_link_libraries(physarum_biology PRIVATE physarum)
target_compile_definitions(physarum_biology PRIVATE PHYSARUM_SOURCE_FINGERPRINT="${ENGINE_SOURCE_FINGERPRINT}")
